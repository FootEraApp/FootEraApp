model Assinatura {
  id        String @id @default(uuid())
  usuarioId String

  contextoKey                  String
  contextoKind                 String?
  contextoTipo                 String?
  contextoPerfilId             String?
  contextoOrganizacaoId        String?
  contextoLegacyOrganizationId String?

  plano                     String
  periodicidade             Periodicidade
  startsAt                  DateTime
  renovaEm                  DateTime
  canceledAt                DateTime?
  ativo                     Boolean          @default(true)
  lembreteEnviado           Boolean          @default(false)
  providerSubscriptionId    String?          @unique
  bloqueadoEm               DateTime?
  metodoPreferido           MetodoPagamento?
  metodoPreferidoDefinidoEm DateTime?
  status                    AssinaturaStatus @default(TRIAL)
  trialEndsAt               DateTime?
  trialStartsAt             DateTime?

  usuario Usuario @relation(fields: [usuarioId], references: [id], onDelete: Cascade)

  @@unique([usuarioId, plano, contextoKey])
  @@index([contextoKey])
  @@index([contextoKind])
  @@index([contextoTipo])
  @@index([contextoPerfilId])
  @@index([contextoOrganizacaoId])
  @@index([contextoLegacyOrganizationId])
  @@index([startsAt])
  @@index([canceledAt])
  @@index([status])
  @@index([trialEndsAt])
}
model BillingComunicacao {
  id           String                   @id @default(uuid())
  assinaturaId String
  cicloKey     String
  evento       String
  canal        BillingComunicacaoCanal
  status       BillingComunicacaoStatus @default(PENDENTE)

  tentativas         Int       @default(0)
  enviadoEm          DateTime?
  proximaTentativaEm DateTime?
  ultimoErro         String?

  criadoEm     DateTime @default(now())
  atualizadoEm DateTime @updatedAt

  @@unique([assinaturaId, cicloKey, evento, canal])
  @@index([status, proximaTentativaEm])
  @@index([assinaturaId])
}
