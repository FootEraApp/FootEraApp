          usuarioId: args.usuarioId,
          pagamentoId: pagamento.id,
          status: "RESERVADA",
          expiraEm: new Date(
            agora.getTime() + CUPOM_RESERVA_REVISAO_MS
          ),
        },
      });

      return pagamento;
  };

  if (args.tx) {
    return executar(args.tx);
  }

  return prisma.$transaction(
    executar,
    {
      maxWait: 10000,
      timeout: 15000,
    }
  );
}

function onlyDigits(s: string) {
  return (s || "").replace(/\D+/g, "");
}

function addMonths(d: Date, months: number) {
  const dt = new Date(d.getTime());
  dt.setMonth(dt.getMonth() + months);
  return dt;
}

function monthsForPeriodicidade(periodicidade?: Periodicidade | null) {
  return periodicidade === "Anual" ? 12 : 1;
}

function diffDays(a: Date, b: Date) {
  const ms = a.getTime() - b.getTime();
  return Math.ceil(ms / (24 * 60 * 60 * 1000));
}

  | "VENCIMENTO_7_DIAS"
  | "BLOQUEIO"
  | "TRIAL_ENCERRADO";

type ComunicacaoBillingArgs = {
  assinaturaId: string;
  usuarioId: string;
  cicloKey: string;
  evento: TipoEventoBilling;
  canal: BillingComunicacaoCanal;
  plano: string;
  contexto: string;
  dataVencimento: Date;
  diasRestantes?: number;
};

const BILLING_MAX_TENTATIVAS = 5;
const BILLING_LEASE_MS = 30 * 60 * 1000;
const BILLING_RETRY_MS = 60 * 60 * 1000;

async function processarComunicacaoBilling(
  args: ComunicacaoBillingArgs
): Promise<boolean> {
  const chave = {
    assinaturaId: args.assinaturaId,
    cicloKey: args.cicloKey,
    evento: args.evento,
    canal: args.canal,
  };

  const registro =
    await prisma.billingComunicacao.upsert({
      where: {
        assinaturaId_cicloKey_evento_canal: chave,
      },
      create: {
        ...chave,
        status: BillingComunicacaoStatus.PENDENTE,
      },
      update: {},
    });

  const agora = new Date();
  const limiteLease = new Date(
    agora.getTime() - BILLING_LEASE_MS
  );

  const adquirido =
    await prisma.billingComunicacao.updateMany({
      where: {
        id: registro.id,
        tentativas: {
          lt: BILLING_MAX_TENTATIVAS,
        },
        OR: [
          {
            status: BillingComunicacaoStatus.PENDENTE,
          },
          {
            status: BillingComunicacaoStatus.FALHOU,
            OR: [
              { proximaTentativaEm: null },
              { proximaTentativaEm: { lte: agora } },
            ],
          },
          {
            status: BillingComunicacaoStatus.PROCESSANDO,
            atualizadoEm: {
              lt: limiteLease,
            },
          },
        ],
      },
      data: {
        status: BillingComunicacaoStatus.PROCESSANDO,
        tentativas: { increment: 1 },
        ultimoErro: null,
        proximaTentativaEm: null,
      },
    });

  // Outra execuÃ§Ã£o jÃ¡ estÃ¡ processando,
  // ou essa comunicaÃ§Ã£o jÃ¡ foi enviada.
  if (adquirido.count === 0) {
    return false;
  }

  try {
    const assinaturaAtual =
      await prisma.assinatura.findUnique({
        where: {
          id: args.assinaturaId,
        },
        select: {
          id: true,
          usuarioId: true,
          status: true,
          ativo: true,
          renovaEm: true,
          trialEndsAt: true,
          bloqueadoEm: true,
        },
      });

    const agoraValidacao = new Date();

    const fimAtual =
      args.evento === "TRIAL_ENCERRADO" ||
      (
        args.evento === "VENCIMENTO_7_DIAS" &&
        assinaturaAtual?.status === "TRIAL"
      )
        ? assinaturaAtual?.trialEndsAt ??
          assinaturaAtual?.renovaEm
        : assinaturaAtual?.renovaEm;

    const mesmoCiclo =
      fimAtual instanceof Date &&
      fimAtual.getTime() ===
        args.dataVencimento.getTime() &&
      args.cicloKey ===
        args.dataVencimento.toISOString();

    const avisoValido =
      args.evento === "VENCIMENTO_7_DIAS" &&
      assinaturaAtual?.ativo === true &&
      (
        assinaturaAtual.status === "ATIVA" ||
        assinaturaAtual.status === "TRIAL"
      ) &&
      args.dataVencimento > agoraValidacao &&
      diffDays(
        args.dataVencimento,
        agoraValidacao
      ) <= 7;

    const bloqueioValido =
      args.evento === "BLOQUEIO" &&
      assinaturaAtual?.status === "BLOQUEADA" &&
      assinaturaAtual.ativo === false &&
      assinaturaAtual.bloqueadoEm !== null &&
      assinaturaAtual.trialEndsAt === null;

    const trialEncerradoValido =
      args.evento === "TRIAL_ENCERRADO" &&
      assinaturaAtual?.status === "BLOQUEADA" &&
      assinaturaAtual.ativo === false &&
      assinaturaAtual.bloqueadoEm !== null &&
      assinaturaAtual.trialEndsAt !== null;

    const comunicacaoValida =
      assinaturaAtual?.usuarioId === args.usuarioId &&
      mesmoCiclo &&
      (
        avisoValido ||
        bloqueioValido ||
        trialEncerradoValido
      );

    if (!comunicacaoValida) {
      // Encerra definitivamente esta tentativa,
      // sem enviar uma mensagem do ciclo antigo.
      await prisma.billingComunicacao.update({
        where: {
          id: registro.id,
        },
        data: {
          status: BillingComunicacaoStatus.FALHOU,
          tentativas: BILLING_MAX_TENTATIVAS,
          ultimoErro:
            "ComunicaÃ§Ã£o descartada: assinatura renovada, ciclo alterado ou status incompatÃ­vel.",
          proximaTentativaEm: null,
        },
      });

      console.log(
        "[billing] ComunicaÃ§Ã£o antiga descartada:",
        {
          assinaturaId: args.assinaturaId,
          evento: args.evento,
          canal: args.canal,
        }
      );

      return false;
    }

    const usuario = await prisma.usuario.findUnique({
      where: {
        id: args.usuarioId,
      },
      select: {
        nome: true,
        email: true,
        deletedAt: true,
      },
    });

    if (!usuario || usuario.deletedAt) {
      throw new Error(
        "UsuÃ¡rio inexistente ou com exclusÃ£o solicitada."
      );
    }

    const titulo =
      args.evento === "VENCIMENTO_7_DIAS"
        ? "Sua assinatura estÃ¡ perto do vencimento"
        : args.evento === "TRIAL_ENCERRADO"
          ? "Seu trial terminou"
          : "Sua assinatura venceu";
    const vencimentoFormatado =
      args.dataVencimento.toLocaleDateString(
        "pt-BR",
        { timeZone: "UTC" }
      );

    const mensagem =
      args.evento === "VENCIMENTO_7_DIAS"
        ? `A assinatura ${args.plano}, vinculada ao ${args.contexto}, vence em ${vencimentoFormatado}. Renove para continuar com seus benefÃ­cios.`
        : args.evento === "TRIAL_ENCERRADO"
          ? `O perÃ­odo grÃ¡tis da assinatura ${args.plano}, vinculada ao ${args.contexto}, terminou em ${vencimentoFormatado}.`
          : `A assinatura ${args.plano}, vinculada ao ${args.contexto}, venceu em ${vencimentoFormatado} e foi bloqueada.`;

    if (
      args.canal === BillingComunicacaoCanal.APP
    ) {
      await criarNotificacaoBilling({
        usuarioId: args.usuarioId,
        tipo:
          args.evento === "VENCIMENTO_7_DIAS"
            ? NotificacaoTipo.BILLING_WARNING
            : NotificacaoTipo.BILLING_BLOCKED,
        titulo,
        mensagem,
        link: "/pagamentos",
      });
    } else {
      if (!usuario.email?.trim()) {
        throw new Error(
          "UsuÃ¡rio sem e-mail cadastrado."
        );
      }

      await sendBillingEmail({
        to: usuario.email,
        nome: usuario.nome,
        plano: args.plano,
        contexto: args.contexto,
        tipo:
          args.evento === "VENCIMENTO_7_DIAS"
            ? "VENCIMENTO"
            : args.evento === "TRIAL_ENCERRADO"
              ? "TRIAL_ENCERRADO"
              : "BLOQUEIO",
        diasRestantes: args.diasRestantes,
        dataVencimento: args.dataVencimento,
      });
    }

    await prisma.billingComunicacao.update({
      where: {
        id: registro.id,
      },
      data: {
        status: BillingComunicacaoStatus.ENVIADO,
        enviadoEm: new Date(),
        ultimoErro: null,
        proximaTentativaEm: null,
      },
    });

    return true;
  } catch (error) {
    const erro =
      error instanceof Error
        ? error.message
        : String(error);

    const proximaTentativaEm = new Date(
      Date.now() + BILLING_RETRY_MS
    );

    await prisma.billingComunicacao.update({
      where: {
        id: registro.id,
      },
      data: {
        status: BillingComunicacaoStatus.FALHOU,
        ultimoErro: erro.slice(0, 1000),
        proximaTentativaEm,
      },
    });

    console.error(
      "[billing] Falha na comunicaÃ§Ã£o:",
      {
        assinaturaId: args.assinaturaId,
        evento: args.evento,
        canal: args.canal,
        erro,
      }
    );

    return false;
  }
}

async function enviarComunicacoesBilling(
  args: Omit<
    ComunicacaoBillingArgs,
    "canal"
  >
) {
  const canais = [
    BillingComunicacaoCanal.APP,
    BillingComunicacaoCanal.EMAIL,
  ];

  const resultados = await Promise.allSettled(
    canais.map((canal) =>
      processarComunicacaoBilling({
        ...args,
        canal,
      })
    )
  );

  for (const resultado of resultados) {
    if (resultado.status === "rejected") {
      console.error(
        "[billing] Erro no processamento de canal:",
        resultado.reason
      );
    }
  }
}

export async function processExpiringSubscriptions(
  options: {

) {
  if (BILLING_CUPOM_RESERVA_QA) {
    const reservasEmRevisao =

    if (reservasEmRevisao > 0) {
      console.log(
        "[billing] Reservas de cupom aguardando conciliaÃ§Ã£o:",
        reservasEmRevisao
      );
    }
  }
  const now = new Date();

  const assinaturas = await prisma.assinatura.findMany({
    where: {
      ...(options.usuarioId
        ? { usuarioId: options.usuarioId }
        : {}),

      ...(options.assinaturaId
        ? { id: options.assinaturaId }
        : {}),

      OR: [
        { status: "TRIAL", ativo: true },
        { status: "ATIVA", ativo: true },
        { status: "CANCELADA", ativo: true },

        // Recupera comunicaÃ§Ãµes de bloqueios anteriores.
        {
          status: "BLOQUEADA",
          bloqueadoEm: { not: null },
        },
      ],
    },
    orderBy: {
      startsAt: "asc",
    },
  });

  let avisadas = 0;
  let bloqueadas = 0;
  let desativadas = 0;

  for (const a of assinaturas) {
    try {
      const dataBase =
        a.status === "TRIAL"
          ? a.trialStartsAt ?? a.startsAt
          : a.startsAt;

      const meses =
        monthsForPeriodicidade(a.periodicidade);

      const dataCalculada = dataBase
        ? addMonths(new Date(dataBase), meses)
        : null;

      const dataRaw =
        a.status === "TRIAL" ||
        (
          a.status === "BLOQUEADA" &&
          a.trialEndsAt
        )
          ? a.trialEndsAt ??
            a.renovaEm ??
            dataCalculada
          : a.renovaEm ?? dataCalculada;

      if (!dataRaw) {
        continue;
      }

      const dataLimite = new Date(dataRaw);

      if (
        Number.isNaN(dataLimite.getTime())
      ) {
        continue;
      }

      // A chave do ciclo nÃ£o depende do momento
      // em que o job Ã© executado.
      const cicloKey =
        dataLimite.toISOString();

      const contextoDescricao =
        descricaoContextoAssinatura(a);

      const dadosComunicacao = {
        assinaturaId: a.id,
        usuarioId: a.usuarioId,
        cicloKey,
        plano: a.plano,
        contexto: contextoDescricao,
        dataVencimento: dataLimite,
      };

      const dias = diffDays(dataLimite, now);

      // 1. Aviso de vencimento
      if (
        a.ativo &&
        (
          a.status === "ATIVA" ||
          a.status === "TRIAL"
        ) &&
        dias >= 0 &&
        dias <= 7 &&
        dataLimite > now
      ) {
        await enviarComunicacoesBilling({
          ...dadosComunicacao,
          evento: "VENCIMENTO_7_DIAS",
          diasRestantes: dias,
        });

        // Campo legado mantido para compatibilidade.
        // A entrega real Ã© controlada por canal.
        const avisoApp =
          await prisma.billingComunicacao.findUnique({
            where: {
              assinaturaId_cicloKey_evento_canal: {
                assinaturaId: a.id,
                cicloKey,
                evento: "VENCIMENTO_7_DIAS",
                canal: BillingComunicacaoCanal.APP,
              },
            },
            select: {
              status: true,
            },
          });

        if (
          avisoApp?.status ===
          BillingComunicacaoStatus.ENVIADO
        ) {
            if (!a.lembreteEnviado) {
              const atualizacao =
                await prisma.assinatura.updateMany({
                  where: {
                    id: a.id,
                    ativo: true,
                    status: a.status,
                    renovaEm: a.renovaEm,
                    trialEndsAt: a.trialEndsAt,
                    lembreteEnviado: false,
                  },
                  data: {
                    lembreteEnviado: true,
                  },
                });

              avisadas += atualizacao.count;
            }
        }
      }

      // 2. Trial encerrado ou assinatura vencida
      if (
        dataLimite <= now &&
        (
          a.status === "TRIAL" ||
          a.status === "ATIVA"
        ) &&
        a.ativo
      ) {
      const evento: TipoEventoBilling =
        a.status === "TRIAL"
          ? "TRIAL_ENCERRADO"
          : "BLOQUEIO";

      const alteracao = await prisma.$transaction(
        async (tx) => {
          const resultado = await tx.assinatura.updateMany({
            where: {
              id: a.id,
              ativo: true,
              status: a.status,
              renovaEm: a.renovaEm,
              ...(a.status === "TRIAL"
                ? { trialEndsAt: a.trialEndsAt }
                : {}),
            },
            data: {
              status: "BLOQUEADA",
              ativo: false,
              canceledAt: now,
              bloqueadoEm: now,
              renovaEm: dataLimite,
              ...(a.status === "TRIAL"
                ? { trialEndsAt: dataLimite }
                : {}),
            },
          });

          if (resultado.count === 0) {
            return resultado;
          }

          await tx.billingComunicacao.createMany({
            data: [
              {
                assinaturaId: a.id,
                cicloKey,
                evento,
                canal: BillingComunicacaoCanal.APP,

                cicloKey,
                evento,
                canal: BillingComunicacaoCanal.EMAIL,

        }

      if (alteracao.count > 0) {

        });
      }

      continue;
      }

      // 3. Recuperar comunicaÃ§Ãµes pendentes de bloqueios
      // jÃ¡ registrados pelo sistema novo.
      if (
        a.status === "BLOQUEADA" &&
        a.bloqueadoEm
      ) {
        const evento: TipoEventoBilling =
          a.trialEndsAt
            ? "TRIAL_ENCERRADO"
            : "BLOQUEIO";

        // SÃ³ recuperamos comunicaÃ§Ãµes que jÃ¡
        // comeÃ§aram a ser processadas.
        // Bloqueios antigos sem registro sÃ£o ignorados.
        const comunicacaoExistente =
          await prisma.billingComunicacao.findFirst({
            where: {
              assinaturaId: a.id,
              cicloKey,
              evento,
            },
            select: {
              id: true,
            },
          });

        if (!comunicacaoExistente) {
          continue;
        }

        // Verifica novamente o estado atual,
        // pois um webhook pode ter renovado a
        // assinatura apÃ³s a consulta inicial.
        const assinaturaAtual =
          await prisma.assinatura.findUnique({
            where: {
              id: a.id,
            },
            select: {
              status: true,
              ativo: true,
