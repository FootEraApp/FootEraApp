model BillingCompensacaoEvento {
  id String @id @default(uuid())

  pagamentoId String @unique
  pendenciaId String @unique

  chaveIdempotencia String @unique

  status BillingCompensacaoEventoStatus @default(PREPARADA)

  valor       Decimal @db.Decimal(10, 2)
  moeda       String  @default("BRL")
  providerRef String

  tentativas Int @default(0)

  ultimoErro   String?
  confirmadoEm DateTime?

  criadoEm     DateTime @default(now())
  atualizadoEm DateTime @updatedAt

  pagamento Pagamento @relation(fields: [pagamentoId], references: [id], onDelete: Restrict)

  pendencia            BillingPendenciaEvento                  @relation(fields: [pendenciaId], references: [id], onDelete: Restrict)
  historicoTentativas  BillingCompensacaoTentativa[]
  consultasRecuperacao BillingCompensacaoConsultaRecuperacao[]

  @@index([status, criadoEm])
}

enum BillingCompensacaoTentativaStatus {
  EM_PROCESSAMENTO
  SIMULACAO_ENCERRADA
  RESULTADO_DESCONHECIDO
  CONFIRMACAO_EXTERNA_PENDENTE
}

model BillingCompensacaoTentativa {
  id            String @id @default(uuid())
  compensacaoId String

  numero Int
  status BillingCompensacaoTentativaStatus @default(EM_PROCESSAMENTO)

  ambiente                 String  @default("QA")
  operacaoExternaExecutada Boolean @default(false)

  estornoProviderRef         String?   @unique
  chaveIdempotenciaOperacao  String?   @unique
  requisicaoEnviadaEm        DateTime?
  postAutorizacaoConsumidaEm DateTime?

  iniciadoEm  DateTime  @default(now())
  encerradoEm DateTime?
  observacao  String?

  compensacao          BillingCompensacaoEvento                @relation(fields: [compensacaoId], references: [id], onDelete: Restrict)
  consultasRecuperacao BillingCompensacaoConsultaRecuperacao[]

  @@unique([compensacaoId, numero])
  @@unique([id, compensacaoId])
  @@index([status, iniciadoEm])
}

model BillingCompensacaoConsultaRecuperacao {
  id            String @id @default(uuid())
  compensacaoId String
  tentativaId   String

  ambiente String  @default("QA")
  simulada Boolean @default(true)

  tipoEvidencia        String
  decisao              String
  motivo               String
  identidadeConfirmada Boolean @default(false)

  providerRef   String
  estornoId     String?
  statusEstorno String?
  valorEstorno  Decimal? @db.Decimal(10, 2)
  moedaEstorno  String?

  consultadoEm DateTime @default(now())

  compensacao BillingCompensacaoEvento    @relation(fields: [compensacaoId], references: [id], onDelete: Restrict)
  tentativa   BillingCompensacaoTentativa @relation(fields: [tentativaId, compensacaoId], references: [id, compensacaoId], onDelete: Restrict)

  @@index([compensacaoId, consultadoEm])
