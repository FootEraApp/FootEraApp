import { PrismaClient } from "@prisma/client";
import { addDays } from "date-fns";
import { sendBillingEmail } from "../utils/mailer.js";

const prisma = new PrismaClient();

const DIAS_AVISO = 3;

export async function processarRenovacoesDiarias() {
  const hoje = new Date();
  const limiteAviso = addDays(hoje, DIAS_AVISO);

  const quaseVencendo = await prisma.assinatura.findMany({
    where: {
      ativo: true,
      renovaEm: {
        gte: hoje,
        lte: limiteAviso,
      },
      lembreteEnviado: false,
    },
    include: { usuario: true },
  });

  for (const ass of quaseVencendo) {
    try {
      await sendBillingEmail({
        to: ass.usuario.email,
        nome: ass.usuario.nome,
        plano: ass.plano,
        contexto: "seu perfil",
        tipo: "VENCIMENTO",
        diasRestantes: DIAS_AVISO,
        dataVencimento: ass.renovaEm,
      });

      await prisma.assinatura.update({
        where: { id: ass.id },
        data: { lembreteEnviado: true },
      });
    } catch (e) {
      console.error("Falha ao enviar e-mail de renovaÃ§Ã£o:", e);
    }
  }

  const vencidas = await prisma.assinatura.findMany({
    where: {
      ativo: true,
      renovaEm: { lt: hoje },
    },
    include: { usuario: true },
  });

  for (const ass of vencidas) {
    const agora = new Date();

    const resultado = await prisma.assinatura.updateMany({
      where: {
        id: ass.id,
        ativo: true,
        renovaEm: {
          lt: agora,
        },
      },
      data: {
        ativo: false,
        canceledAt: agora,
        status: "BLOQUEADA",
        bloqueadoEm: agora,
      },
    });

    if (resultado.count === 0) {
      continue;
    }

    try {
      await sendBillingEmail({
        to: ass.usuario.email,
        nome: ass.usuario.nome,
        plano: ass.plano,
        contexto: "seu perfil",
        tipo: "BLOQUEIO",
        dataVencimento: ass.renovaEm,
      });
    } catch (e) {
      console.error("Falha ao informar expiraÃ§Ã£o:", e);
    }
  }
}
