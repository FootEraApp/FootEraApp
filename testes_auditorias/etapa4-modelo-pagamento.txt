model Pagamento {
  id                           String                    @id @default(uuid())
  usuarioId                    String
  contextoKind                 String?
  contextoTipo                 String?
  contextoPerfilId             String?
  contextoOrganizacaoId        String?
  contextoLegacyOrganizationId String?
  plano                        String
  periodicidade                Periodicidade
  metodo                       MetodoPagamento
  status                       PagamentoStatus           @default(PENDENTE)
  valor                        Decimal                   @db.Decimal(10, 2)
  moeda                        String                    @default("BRL")
  cupomId                      String?
  provider                     String?
  providerRef                  String?
  criadoEm                     DateTime                  @default(now())
  pagoEm                       DateTime?
  canceladoEm                  DateTime?
  reembolsadoEm                DateTime?
  updatedAt                    DateTime                  @updatedAt
  pagadorNome                  String?
  pagadorEmail                 String?
  pagadorCpf                   String?
  pagadorTelefone              String?
  cardBrand                    String?
  cardLast4                    String?
  cardExpMes                   Int?
  cardExpAno                   Int?
  pixCopiaECola                String?
  boletoLinhaDigitavel         String?
  meta                         Json?
  CupomResgate                 CupomResgate[]
  cupom                        Cupom?                    @relation(fields: [cupomId], references: [id], onDelete: SetNull)
  usuario                      Usuario                   @relation(fields: [usuarioId], references: [id], onDelete: Cascade)
  beneficios                   PagamentoBeneficio[]
  creatorVendas                CreatorVenda[]
  reservasCupom                CupomReserva[]
  pendenciaEvento              BillingPendenciaEvento?
  compensacaoEvento            BillingCompensacaoEvento?
  operacaoCheckout             BillingCheckoutOperacao?

  @@unique([provider, providerRef])
  @@index([usuarioId, criadoEm])
  @@index([status])
  @@index([status, criadoEm])
  @@index([contextoKind])
  @@index([contextoTipo])
  @@index([contextoPerfilId])
  @@index([contextoOrganizacaoId])
  @@index([contextoLegacyOrganizationId])
}

enum BillingCheckoutOperacaoStatus {
  PREPARADA
  ENVIO_POSSIVEL
  CONFIRMADA
  INCONCLUSIVA
  FALHA_COMPROVADA
}

model BillingCheckoutOperacao {
  id            String                        @id @default(uuid())
  pagamentoId   String                        @unique
  usuarioId     String
  chaveCliente  String
  chaveOperacao String                        @unique
  modalidade    String
  status        BillingCheckoutOperacaoStatus @default(PREPARADA)

  fingerprintRequisicao String
  providerRef           String?
  ultimoErro            String?

  envioPossivelEm DateTime?
  confirmadoEm    DateTime?
  criadoEm        DateTime  @default(now())
  atualizadoEm    DateTime  @updatedAt

  pagamento Pagamento @relation(fields: [pagamentoId], references: [id], onDelete: Restrict)

  @@unique([usuarioId, chaveCliente])
  @@index([status, criadoEm])
  @@index([usuarioId, criadoEm])
}

enum BillingPendenciaEventoStatus {
  PENDENTE_ANALISE
  EM_ANALISE
  AGUARDANDO_ESTORNO
  ESTORNO_CONFIRMADO
  RESOLVIDA_MANUALMENTE
}

model BillingPendenciaEvento {
  id          String @id @default(uuid())
  pagamentoId String @unique
  eventoId    String

  motivo String
  status BillingPendenciaEventoStatus @default(PENDENTE_ANALISE)

  providerRef String
  valor       Decimal @db.Decimal(10, 2)
  moeda       String  @default("BRL")

  observacao   String?
  criadoEm     DateTime  @default(now())
  atualizadoEm DateTime  @updatedAt
