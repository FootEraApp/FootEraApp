
import dotenv from "dotenv";

// Execute o script a partir da raiz do projeto.
dotenv.config({ path: "server/.env" });

async function main() {
  const databaseUrl = process.env.DATABASE_URL;
  const usuarioId = process.env.BILLING_TEST_USER_ID?.trim();
  const assinaturaId =
    process.env.BILLING_TEST_ASSINATURA_ID?.trim();

    if (!assinaturaId) {
    throw new Error(
        "Defina BILLING_TEST_ASSINATURA_ID."
    );
    }

  if (
    process.env.BILLING_TEST_MODE !== "true"
  ) {
    throw new Error(
      "Ative BILLING_TEST_MODE=true para este teste."
    );
  }

  if (!databaseUrl) {
    throw new Error("DATABASE_URL não configurada.");
  }

  const url = new URL(databaseUrl);

  if (
    !["127.0.0.1", "localhost"].includes(
      url.hostname
    ) ||
    url.port !== "5433"
  ) {
    throw new Error(
      "Teste bloqueado: DATABASE_URL não aponta para o PostgreSQL local na porta 5433."
    );
  }

  if (process.env.SMTP_HOST?.trim()) {
    throw new Error(
      "Teste bloqueado: SMTP_HOST configurado. Remova essa variável do ambiente de teste para utilizar o Ethereal."
    );
  }

  if (
    process.env.ENABLE_BILLING_SCHEDULER === "true" ||
    process.env.ENABLE_ACCOUNT_PURGE === "true"
  ) {
    throw new Error(
      "Desative os schedulers antes do teste."
    );
  }

  if (!usuarioId) {
    throw new Error(
      "Defina BILLING_TEST_USER_ID."
    );
  }

  const { prisma } = await import("../prisma.js");

  try {
    const usuario = await prisma.usuario.findUnique({
      where: { id: usuarioId },
      select: {
        id: true,
        nome: true,
        email: true,
        deletedAt: true,
      },
    });

    if (!usuario || usuario.deletedAt) {
      throw new Error(
        "Usuário de teste inexistente ou excluído."
      );
    }

    const assinaturas = await prisma.assinatura.findMany({
      where: {
        id: assinaturaId,
        usuarioId,
        },
      select: {
        id: true,
        plano: true,
        status: true,
        ativo: true,
        renovaEm: true,
      },
    });

    if (assinaturas.length !== 1) {
      throw new Error(
        "O usuário de teste deve possuir exatamente uma assinatura para este primeiro teste."
      );
    }

    console.log("[billing-test] Usuário:", usuario.nome);
    console.log("[billing-test] Assinatura:", assinaturas);

    const { processExpiringSubscriptions } =
      await import("../controllers/billingController.js");

    const resultado =
        await processExpiringSubscriptions({
            usuarioId,
            assinaturaId,
        });

    console.log("[billing-test] Resultado:", resultado);

    const assinaturasDepois =
      await prisma.assinatura.findMany({
        where: {
        id: assinaturaId,
        usuarioId,
        },
        select: {
          id: true,
          status: true,
          ativo: true,
          lembreteEnviado: true,
          bloqueadoEm: true,
        },
      });

    const comunicacoes =
      await prisma.billingComunicacao.findMany({
        where: {
          assinaturaId: {
            in: assinaturas.map((a) => a.id),
          },
        },
        orderBy: {
          criadoEm: "asc",
        },
      });

    const notificacoes =
      await prisma.notificacao.findMany({
        where: {
          usuarioId,
          tipo: {
            in: [
              "BILLING_WARNING",
              "BILLING_BLOCKED",
            ],
          },
        },
        select: {
          id: true,
          titulo: true,
          mensagem: true,
          lida: true,
        },
        orderBy: {
          createdAt: "desc",
        },
        take: 10,
      });

    console.log(
      "[billing-test] Assinaturas após:",
      assinaturasDepois
    );

    console.table(
      comunicacoes.map((item) => ({
        evento: item.evento,
        canal: item.canal,
        status: item.status,
        tentativas: item.tentativas,
        erro: item.ultimoErro,
      }))
