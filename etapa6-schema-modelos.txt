model Assinatura {
  id        String @id @default(uuid())
  usuarioId String

  contextoKey                  String
  contextoKind                 String?
  contextoTipo                 String?
  contextoPerfilId             String?
  contextoOrganizacaoId        String?
  contextoLegacyOrganizationId String?

  plano                     String
  periodicidade             Periodicidade
  startsAt                  DateTime
  renovaEm                  DateTime
  canceledAt                DateTime?
  ativo                     Boolean          @default(true)
  lembreteEnviado           Boolean          @default(false)
  providerSubscriptionId    String?          @unique
  bloqueadoEm               DateTime?
  metodoPreferido           MetodoPagamento?
  metodoPreferidoDefinidoEm DateTime?
  status                    AssinaturaStatus @default(TRIAL)
  trialEndsAt               DateTime?
  trialStartsAt             DateTime?

  usuario Usuario @relation(fields: [usuarioId], references: [id], onDelete: Cascade)

  @@unique([usuarioId, plano, contextoKey])
  @@index([contextoKey])
  @@index([contextoKind])
  @@index([contextoTipo])
  @@index([contextoPerfilId])
  @@index([contextoOrganizacaoId])
  @@index([contextoLegacyOrganizationId])
  @@index([startsAt])
  @@index([canceledAt])
  @@index([status])
  @@index([trialEndsAt])
}

model Notificacao {
  id        String          @id @default(uuid())
  usuarioId String
  titulo    String
  mensagem  String
  link      String?
  lida      Boolean         @default(false)
  createdAt DateTime        @default(now())
  actorId   String?
  tipo      NotificacaoTipo @default(GENERICA)
  actor     Usuario?        @relation("NotificacaoActor", fields: [actorId], references: [id], onDelete: SetNull)
  usuario   Usuario         @relation("NotificacaoUsuario", fields: [usuarioId], references: [id], onDelete: Cascade)

  @@index([usuarioId, createdAt])
  @@index([actorId])

model Conquista {
  id        String               @id @default(uuid())
  codigo    String               @unique
  titulo    String
  descricao String?
  tipo      ConquistaTipo        @default(GERAL)
  publico   ConquistaOwnerTipo[] @default([Atleta])
  icon      String?
  iconUrl   String?
  pontos    Int?                 @default(0)
  meta      Int?
  ativo     Boolean              @default(true)
  createdAt DateTime             @default(now())
  updatedAt DateTime             @updatedAt
  vinculos  ConquistaVinculo[]

  @@index([tipo])
  @@index([ativo])
}

model ConquistaVinculo {
  id                String             @id @default(uuid())
  conquistaId       String
  ownerTipo         ConquistaOwnerTipo
  ownerId           String
  atletaId          String?
  professorId       String?
  clubeId           String?
  escolinhaId       String?
  conquistadoEm     DateTime?
  progresso         Int                @default(0)
  concluida         Boolean            @default(false)
  refTipo           String?
  refId             String?
  createdAt         DateTime           @default(now())
  updatedAt         DateTime           @updatedAt
  federacaoId       String?
  learningProfileId String?
  marcaId           String?
  atleta            Atleta?            @relation("ConquistaAtleta", fields: [atletaId], references: [id], onDelete: Cascade)
  clube             Clube?             @relation("ConquistaClube", fields: [clubeId], references: [id], onDelete: Cascade)
  conquista         Conquista          @relation(fields: [conquistaId], references: [id], onDelete: Cascade)
  escolinha         Escolinha?         @relation("ConquistaEscolinha", fields: [escolinhaId], references: [id], onDelete: Cascade)
  federacao         Federacao?         @relation(fields: [federacaoId], references: [id], onDelete: SetNull)
  learningProfile   LearningProfile?   @relation(fields: [learningProfileId], references: [id], onDelete: SetNull)
  marca             Marca?             @relation(fields: [marcaId], references: [id], onDelete: SetNull)
  professor         Professor?         @relation("ConquistaProfessor", fields: [professorId], references: [id], onDelete: Cascade)

  @@unique([ownerTipo, ownerId, conquistaId])
  @@index([ownerTipo, ownerId])
  @@index([atletaId])
  @@index([professorId])
  @@index([clubeId])
  @@index([escolinhaId])
  @@index([concluida, conquistadoEm])
}
